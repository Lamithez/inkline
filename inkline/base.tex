% 独立段落注释：直角浅灰底、衬线文字与贴齐左缘的深蓝线。
\newsavebox{\ink@notebox}
\newlength{\ink@noteheight}
\newcommand{\inknote}[1]{%
  \par\smallskip\begingroup
  \setlength{\parskip}{0pt}%
  \noindent\begin{minipage}{\linewidth}
    \small\rmfamily\mdseries\upshape
    % sbox 为测量内容建立正确的颜色组，防止装箱时颜色泄漏。
    \sbox{\ink@notebox}{%
      \begin{minipage}{\dimexpr\linewidth-1.1em-0.8pt\relax}
        \raggedright\setlength{\parskip}{0.2em}%
        \usebeamercolor[fg]{annotation}\noindent#1\par
      \end{minipage}}%
    \setlength{\ink@noteheight}{\dimexpr\ht\ink@notebox+\dp\ink@notebox+0.9em\relax}%
    \usebeamercolor{annotation}%
    \begin{pgfpicture}
      \pgfsetcornersarced{\pgfpointorigin}%
      \pgfsetfillcolor{bg}%
      \pgfpathrectangle{\pgfpointorigin}{\pgfpoint{\linewidth}{\ink@noteheight}}%
      \pgfusepath{fill}%
      \pgfsetfillcolor{AcademicBlue}%
      \pgfpathrectangle{\pgfpointorigin}{\pgfpoint{0.8pt}{\ink@noteheight}}%
      \pgfusepath{fill}%
      \pgftext[left,bottom,at={\pgfpoint{\dimexpr0.55em+0.8pt\relax}{0.45em}}]{\usebox{\ink@notebox}}%
    \end{pgfpicture}%
  \end{minipage}\par\endgroup\smallskip}
