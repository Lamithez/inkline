% =============================================================================
% main.tex —— 唯一编译入口
% -----------------------------------------------------------------------------
% 编译：tectonic main.tex      （生成 main.pdf）
%
% 加载顺序不可随意调换：
%   1) 先读 style-*.tex：它用 \PassOptionsToClass 传字号，并提供
%      \SlideGeometry / \ApplySlideStyle / \InkMargin / \SourceNoteSize，
%      其中 \SlideGeometry 必须在 \documentclass 之后立刻执行。
%   2) 再读 config/ 下的公共设置：base → colors → layout → blocks → commands。
%   3) 最后 \ApplySlideStyle 施加版式字号与页脚，再读封面元信息。
% =============================================================================

% -----------------------------------------------------------------------------
% 1. 选择版式：small（默认，16 × 9 cm）或 large（24 × 13.5 cm）
%    只改下面这一行即可切换；两版共用正文、字体、配色与图片。
% -----------------------------------------------------------------------------
\input{config/style-small}
\documentclass[aspectratio=169,t]{beamer}
\SlideGeometry

% -----------------------------------------------------------------------------
% 2. 导言区（公共设置，与版式无关）
% -----------------------------------------------------------------------------
\input{config/base}       % 宏包、中文字体、表格、文献、正文命令
\input{config/colors}     % 墨线配色与 Block 语义色
\input{config/layout}     % 页标题、边距、页脚与页码、过渡页、目录模板
\input{config/blocks}     % 三色圆角 Block
\input{config/commands}   % 目录与脚注字体
\ApplySlideStyle          % 本版式的字号与页脚
\input{config/metadata}   % 标题与日期（封面字段）

% 正文段间距；目录页内单独归零。
\setlength{\parskip}{0.5em}
\begin{document}

% -----------------------------------------------------------------------------
% 3. 封面
%    局部分组加一段脚注后的固定留白：Beamer 把脚注排在底部，
%    plain 封面容易贴边，把脚注文字上移 0.4cm。分组保证正文页脚注不受影响。
% -----------------------------------------------------------------------------
\begingroup
\addtobeamertemplate{footnote}{}{\par\vspace*{0.4cm}}
\begin{frame}[plain,c,noframenumbering]
  \titlepage
\end{frame}
\endgroup

% -----------------------------------------------------------------------------
% 4. 目录（不计入页码）
% -----------------------------------------------------------------------------
\begin{frame}[plain,noframenumbering]{目录}
  \setlength{\parskip}{0pt}
  \tableofcontents
\end{frame}

% -----------------------------------------------------------------------------
% 5. 正文：研究背景 → 调研结果 → 分析与讨论 → 结论
% -----------------------------------------------------------------------------
\input{sections/01-introduction}
\input{sections/02-survey}
\input{sections/03-analysis}
\input{sections/04-conclusion}

% -----------------------------------------------------------------------------
% 6. 示例页（可选）
%    05-examples.tex 汇总了 Block、分栏、图文、表格、公式与分步显示的模板页，
%    方便复制到自己的章节。不需要时删除下面这一行即可。
% -----------------------------------------------------------------------------
\input{sections/05-examples}

% -----------------------------------------------------------------------------
% 7. 附录（可选）
%    \appendix 之后页码切换为独立的 A-1 / A-3 编号（见 config/layout.tex）。
%    参考文献与补充材料都在 90-appendix.tex；不需要附录时删除下面两行。
% -----------------------------------------------------------------------------
\appendix
\input{sections/90-appendix}

\end{document}
