\section{研究背景}

\subsection{问题背景}

\begin{frame}{研究背景}
  \begin{keypoints}
    \item 用一至两句话说明研究问题及其应用场景。
    \item 交代本次调研的范围、对象和时间边界。
    \item 明确后续分析所采用的关键定义。
  \end{keypoints}
\end{frame}

\subsection{研究问题}

\begin{frame}{研究问题}
  \begin{block}{核心问题}
    在此处写出本次汇报希望回答的主要问题。
  \end{block}
  \vspace{0.6em}
  \begin{enumerate}
    \item \alert{子问题 A}：描述对象、指标或机制。
    \item 子问题 B：说明比较维度或评价标准。
  \end{enumerate}
\end{frame}
