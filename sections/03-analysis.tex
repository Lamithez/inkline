\begin{frame}{关键发现}
  \begin{enumerate}
    \item \keypoint{发现一：}写明证据支持的具体结论。
    \item \keypoint{发现二：}说明不同资料之间的一致或冲突之处。
    \item \keypoint{发现三：}指出结论适用的条件和边界。
  \end{enumerate}
\end{frame}

\begin{frame}{局限性}
  \begin{enumerate}
    \item 数据覆盖范围可能限制结论的外推。

    \item 部分结论仍需实验或更多数据验证。
  \end{enumerate}
